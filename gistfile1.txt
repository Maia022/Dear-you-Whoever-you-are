        function toggleAudio() {
            const btn = document.getElementById('soundToggleBtn');
            const icon = document.getElementById('soundIcon');
            const label = document.getElementById('soundLabel');

            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }

            if (!state.audioPlaying) {
                // Play soft atmospheric synth drone
                audioCtx.resume();
                
                ambientOscillator = audioCtx.createOscillator();
                ambientGain = audioCtx.createGain();

                ambientOscillator.type = 'sine';
                ambientOscillator.frequency.setValueAtTime(110, audioCtx.currentTime); // A2 soft tone
                
                // Lowpass filter for warm atmosphere
                const filter = audioCtx.createBiquadFilter();
                filter.type = 'lowpass';
                filter.frequency.setValueAtTime(350, audioCtx.currentTime);

                ambientGain.gain.setValueAtTime(0.01, audioCtx.currentTime);
                ambientGain.gain.exponentialRampToValueAtTime(0.08, audioCtx.currentTime + 3);

                ambientOscillator.connect(filter);
                filter.connect(ambientGain);
                ambientGain.connect(audioCtx.destination);

                ambientOscillator.start();

                state.audioPlaying = true;
                icon.className = "fa-solid fa-volume-high text-amber-300 animate-pulse";
                label.innerText = "Ambient Playing";
                btn.classList.add('border-amber-400/40');
                showToast("Soft atmospheric audio enabled.");
            } else {
                if (ambientGain) {
                    ambientGain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 1);
                    setTimeout(() => {
                        ambientOscillator.stop();
                        ambientOscillator.disconnect();
                    }, 1000);
                }
                state.audioPlaying = false;
                icon.className = "fa-solid fa-volume-xmark text-slate-400";
                label.innerText = "Ambient Audio";
                btn.classList.remove('border-amber-400/40');
            }
        }

        function initStars() {
            const canvas = document.getElementById('starCanvas');
            const ctx = canvas.getContext('2d');
            let width = (canvas.width = window.innerWidth);
            let height = (canvas.height = window.innerHeight);

            window.addEventListener('resize', () => {
                width = canvas.width = window.innerWidth;
                height = canvas.height = window.innerHeight;
            });

            const particles = Array.from({ length: 70 }, () => ({
                x: Math.random() * width,
                y: Math.random() * height,
                radius: Math.random() * 1.5 + 0.5,
                alpha: Math.random() * 0.7 + 0.2,
                speed: Math.random() * 0.15 + 0.05,
                drift: (Math.random() - 0.5) * 0.1
            }));

            function animate() {
                ctx.clearRect(0, 0, width, height);

                particles.forEach(p => {
                    p.y -= p.speed;
                    p.x += p.drift;

                    if (p.y < 0) p.y = height;
                    if (p.x < 0) p.x = width;
                    if (p.x > width) p.x = 0;

                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                    ctx.fillStyle = `rgba(253, 230, 138, ${p.alpha})`;
                    ctx.fill();
                });

                requestAnimationFrame(animate);
            }

            animate();
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/[&<>"']/g, function(m) {
                return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m];
            });

        function loadLocalStorage() {
            // Load public/community letters or default
            const localPublic = localStorage.getItem('dearyou_letters');
            if (localPublic) {
                try { state.letters = JSON.parse(localPublic); } catch (e) { state.letters = DEFAULT_LETTERS; }
            } else {
                state.letters = DEFAULT_LETTERS;
                localStorage.setItem('dearyou_letters', JSON.stringify(DEFAULT_LETTERS));
            }

            // Load private letters
            const localPrivate = localStorage.getItem('dearyou_private_letters');
            if (localPrivate) {
                try { state.privateLetters = JSON.parse(localPrivate); } catch (e) { state.privateLetters = []; }
            }

            // Load saved favorite IDs
            const localFavs = localStorage.getItem('dearyou_favorites');
            if (localFavs) {
                try { state.savedFavorites = JSON.parse(localFavs); } catch (e) { state.savedFavorites = []; }
            }

            updateSavedBadge();
        }

        function handleReleaseLetter(event) {
            event.preventDefault();
            const recipient = document.getElementById('recipientInput').value.trim();
            const title = document.getElementById('titleInput').value.trim() || 'Untitled Whisper';
            const body = document.getElementById('bodyInput').value.trim();
            const privacy = document.getElementById('privacyInput').value;

            if (!recipient || !body) return;

            const newLetter = {
                id: 'l-' + Date.now(),
                recipient: recipient,
                title: title,
                body: body,
                mood: state.composerSelectedMood || 'Hopeful',
                timestamp: `Written at ${new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})} under the midnight sky`,
                reactions: { heard: 1, light: 1, feel: 0, space: 0 },
                isPublic: privacy === 'public'
            };

            const card = document.getElementById('composerCard');
            card.classList.add('releasing');

            setTimeout(() => {
                if (privacy === 'public') {
                    state.letters.unshift(newLetter);
                    localStorage.setItem('dearyou_letters', JSON.stringify(state.letters));
                    showToast("Your letter has been released into the night sky.");
                } else {
                    state.privateLetters.unshift(newLetter);
                    localStorage.setItem('dearyou_private_letters', JSON.stringify(state.privateLetters));
                    showToast("Saved to your Private Sanctuary.");
                }

                clearComposer();
                card.classList.remove('releasing');
                renderFeed();
                renderSanctuary();
                switchTab(privacy === 'public' ? 'feed' : 'sanctuary');
            }, 700);
        }

        function renderFeed() {

            grid.innerHTML = '';

            const filtered = state.letters.filter(letter => {
                const matchesMood = state.selectedMoodFilter === 'all' || letter.mood === state.selectedMoodFilter;
                const matchesSearch = letter.recipient.toLowerCase().includes(state.searchQuery) ||
                                      letter.title.toLowerCase().includes(state.searchQuery) ||
                                      letter.body.toLowerCase().includes(state.searchQuery);
                return matchesMood && matchesSearch;
            });

            if (filtered.length === 0) {
                empty.classList.remove('hidden');
                return;

                card.innerHTML = `

                            <span class="font-serif italic text-amber-200/90 text-sm sm:text-base font-normal tracking-wide">${escapeHtml(letter.recipient)}</span>

        function renderModalReactions(letter) {
            const container = document.getElementById('modalReactionsContainer');
            container.innerHTML = '';

                { key: 'heard', label: 'You are heard', icon: '🕊️' },
                { key: 'light', label: 'Sending light', icon: '✨' },
                { key: 'feel', label: 'I feel this too', icon: '🌊' },
                { key: 'space', label: 'Holding space', icon: '🕯️' }

            reactions.forEach(r => {
                const count = (letter.reactions && letter.reactions[r.key]) || 0;
                const btn = document.createElement('button');
                btn.className = "px-3 py-1.5 rounded-full bg-black/5 dark:bg-white/10 hover:bg-amber-400/20 text-slate-700 dark:text-slate-200 text-xs font-sans transition-all flex items-center gap-1.5 border border-black/10 dark:border-white/10";

                btn.innerHTML = `<span>${r.icon}</span> <span>${r.label}</span> <span class="text-[10px] opacity-70 bg-black/10 dark:bg-white/10 px-1.5 py-0.5 rounded-full">${count}</span>`;

        function sendReaction(letterId, reactionKey) {
            const letter = state.letters.find(l => l.id === letterId);
            if (!letter) return;
            if (!letter.reactions) letter.reactions = {};
            letter.reactions[reactionKey] = (letter.reactions[reactionKey] || 0) + 1;

            localStorage.setItem('dearyou_letters', JSON.stringify(state.letters));
            renderModalReactions(letter);
            renderFeed();
            showToast("Warmth sent anonymously.");
        }

        function toggleAudio() {
            const btn = document.getElementById('soundToggleBtn');
            const icon = document.getElementById('soundIcon');
            const label = document.getElementById('soundLabel');

            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }

            if (!state.audioPlaying) {
                // Play soft atmospheric synth drone
                audioCtx.resume();
                
                ambientOscillator = audioCtx.createOscillator();
                ambientGain = audioCtx.createGain();

                ambientOscillator.type = 'sine';
                ambientOscillator.frequency.setValueAtTime(110, audioCtx.currentTime); // A2 soft tone
                
                // Lowpass filter for warm atmosphere
                const filter = audioCtx.createBiquadFilter();
                filter.type = 'lowpass';
                filter.frequency.setValueAtTime(350, audioCtx.currentTime);

                ambientGain.gain.setValueAtTime(0.01, audioCtx.currentTime);
                ambientGain.gain.exponentialRampToValueAtTime(0.08, audioCtx.currentTime + 3);

                ambientOscillator.connect(filter);
                filter.connect(ambientGain);
                ambientGain.connect(audioCtx.destination);

                ambientOscillator.start();

                state.audioPlaying = true;
                icon.className = "fa-solid fa-volume-high text-amber-300 animate-pulse";
                label.innerText = "Ambient Playing";
                btn.classList.add('border-amber-400/40');
                showToast("Soft atmospheric audio enabled.");
            } else {
                if (ambientGain) {
                    ambientGain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 1);
                    setTimeout(() => {
                        ambientOscillator.stop();
                        ambientOscillator.disconnect();
                    }, 1000);
                }
                state.audioPlaying = false;
                icon.className = "fa-solid fa-volume-xmark text-slate-400";
                label.innerText = "Ambient Audio";
                btn.classList.remove('border-amber-400/40');
            }
        }

        function initStars() {
            const canvas = document.getElementById('starCanvas');
            const ctx = canvas.getContext('2d');
            let width = (canvas.width = window.innerWidth);
            let height = (canvas.height = window.innerHeight);

            window.addEventListener('resize', () => {
                width = canvas.width = window.innerWidth;
                height = canvas.height = window.innerHeight;
            });

            const particles = Array.from({ length: 70 }, () => ({
                x: Math.random() * width,
                y: Math.random() * height,
                radius: Math.random() * 1.5 + 0.5,
                alpha: Math.random() * 0.7 + 0.2,
                speed: Math.random() * 0.15 + 0.05,
                drift: (Math.random() - 0.5) * 0.1
            }));

            function animate() {
                ctx.clearRect(0, 0, width, height);

                particles.forEach(p => {
                    p.y -= p.speed;
                    p.x += p.drift;

                    if (p.y < 0) p.y = height;
                    if (p.x < 0) p.x = width;
                    if (p.x > width) p.x = 0;

                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                    ctx.fillStyle = `rgba(253, 230, 138, ${p.alpha})`;
                    ctx.fill();
                });

                requestAnimationFrame(animate);
            }

            animate();
        }

        function getMoodEmoji(mood) {
            const map = {
                'Hopeful': '🌱',
                'Heartbroken': '🌧️',
                'Nostalgic': '🍂',
                'Grateful': '✨',
                'Silent Longing': '🌙',
                'Confused': '🌫️'
            };
            return map[mood] || '🕊️';
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/[&<>"']/g, function(m) {
                return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m];
            });
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toastMessage');
            toastMsg.innerText = message;
            toast.classList.remove('translate-y-20', 'opacity-0');

                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }
